  <script>
    const API_URL = "http://127.0.0.1:8000/translate";
    const chat = document.getElementById("chat");

    function autoHeight(elem) {
      elem.style.height = "1px";
      elem.style.height = (elem.scrollHeight) + "px";
    }

    function addMessage(text, sender) {
      const row = document.createElement("div");
      row.className = `row ${sender}`;
      const bubble = document.createElement("div");
      bubble.className = "bubble";
      bubble.innerText = text;
      row.appendChild(bubble);
      chat.appendChild(row);
      chat.scrollTop = chat.scrollHeight;
    }

    async function anyToEnglish() {
      const text = document.getElementById("inputText").value.trim();
      if (!text) return;
      addMessage(text, "user");
      document.getElementById("inputText").value = "";
      addMessage("Processing...", "bot");

      try {
        const res = await fetch(API_URL, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ text, mode: "any_to_en" })
        });
        const data = await res.json();
        chat.lastChild.querySelector('.bubble').innerText = `[${data.detected_lang}] Result:\n\n${data.output}`;
      } catch (e) {
        chat.lastChild.querySelector('.bubble').innerText = "Connection Error.";
      }
    }

    async function englishToLang() {
      const text = document.getElementById("inputText").value.trim();
      const lang = document.getElementById("targetLang").value;
      if (!text) return;
      addMessage(text, "user");
      document.getElementById("inputText").value = "";
      addMessage("Translating...", "bot");

      try {
        const res = await fetch(API_URL, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ text, mode: "en_to_lang", target_lang: lang })
        });
        const data = await res.json();
        chat.lastChild.querySelector('.bubble').innerText = data.output;
      } catch (e) {
        chat.lastChild.querySelector('.bubble').innerText = "Connection Error.";
      }
    }
  </script>
